package sftp_test

// Reproduces the shutdown hang on UNMODIFIED litestream v0.5.17 (ccd326c).
//
// Copy this file and testserver_test.go.txt into sftp/ of a clean v0.5.17
// checkout, renaming both to *_test.go, then:
//
//	go test -count=1 -timeout 25s -run TestHangRepro ./sftp/
//
// Expected: the test binary panics at its timeout and dumps every goroutine.
// The write is parked in pkg/sftp clientConn.sendPacket despite the two-second
// context deadline below, because the call is waiting on a reply packet and
// observes no context.

import (
	"bytes"
	"context"
	"path/filepath"
	"testing"
	"time"

	"github.com/superfly/ltx"

	"github.com/benbjohnson/litestream/sftp"
)

func TestHangRepro(t *testing.T) {
	srv := startTestServer(t, stallAtWrite)

	c := sftp.NewReplicaClient()
	c.Host = srv.addr
	c.User = "test"
	c.KeyPath = writeClientKey(t, t.TempDir())
	c.HostKey = srv.hostKey
	c.Path = filepath.Join(srv.root, "upload")

	var buf bytes.Buffer
	enc, err := ltx.NewEncoder(&buf)
	if err != nil {
		t.Fatal(err)
	}
	if err := enc.EncodeHeader(ltx.Header{
		Version: ltx.Version, Flags: ltx.HeaderFlagNoChecksum, PageSize: 4096,
		Commit: 1, MinTXID: 1, MaxTXID: 1, Timestamp: time.Now().UnixMilli(),
	}); err != nil {
		t.Fatal(err)
	}
	if err := enc.EncodePage(ltx.PageHeader{Pgno: 1}, make([]byte, 4096)); err != nil {
		t.Fatal(err)
	}
	if err := enc.Close(); err != nil {
		t.Fatal(err)
	}

	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
	defer cancel()
	_, _ = c.WriteLTXFile(ctx, 0, 1, 1, &buf)
	t.Fatal("the write returned; the hang did not reproduce")
}
